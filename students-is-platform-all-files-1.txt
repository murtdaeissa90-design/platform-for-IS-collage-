# منصة Students IS التعليمية — Next.js

جامعة البيان · كلية علوم الحاسوب وتقانة المعلومات · قسم نظم المعلومات

مبنية بـ **Next.js 14** (تصدير ثابت Static Export) + **Supabase** (تسجيل الدخول والتخزين) + استضافة **Netlify**.

## ⚠️ الإعداد لأول مرة (بالترتيب)

1. **Supabase ← SQL Editor**: افتح `supabase/schema.sql`، غيّر `ADMIN_EMAIL_HERE` ببريد حساب الأدمن الحالي، ثم شغّله كاملاً.
   - بدون هذه الخطوة لن تعمل صفحة الأدمن، وصلاحيات الطلاب لن تكون محمية.
   - بعد التشغيل راجع السياسات القديمة (التعليمات في آخر الملف).
2. **Supabase ← Authentication ← Providers**: تأكد أن Email مفعّل. في **URL Configuration** ضع رابط موقعك على Netlify في `Site URL` وأضفه إلى `Redirect URLs` (لرابط استعادة كلمة المرور).
   - إن أردت أن يدخل الطالب مباشرة بعد التسجيل، عطّل "Confirm email".
3. **GitHub**: ارفع محتويات هذا المجلد إلى المستودع المربوط بـ Netlify (استبدل الملفات القديمة).
4. **Netlify ← Site configuration ← Build & deploy**:
   - Build command: `npm run build`
   - Publish directory: `out`
   - (الملف `netlify.toml` يضبطهما تلقائياً، فإن ظهر تعارض امسح القيم القديمة من لوحة Netlify.)
5. (اختياري) ضع متغيرات البيئة من `.env.example` في Netlify. إن لم تضعها يستخدم المشروع القيم الحالية.

## التعديل من GitHub (يظهر مباشرة في المنصة)

كل `commit` على المستودع يُنشر تلقائياً على Netlify خلال دقائق. الملفات التي تُعدَّل عادةً:

| الملف | ما تغيّره فيه |
|---|---|
| `src/data/semesters.js` | إضافة مادة أو سمستر |
| `src/data/files.js` | روابط المحاضرات والشيتات والامتحانات |
| `src/data/questions.js` | أسئلة الاختيار من متعدد |
| `src/data/aliases.js` | أسماء عربية بديلة للمواد (للبحث) |

## الخطوط

- English: **Inter** (الأساسي) · **Sora** (العناوين) · **JetBrains Mono** (الأكواد والأرقام)
- العربي: **Amiri Bold** (العناوين) · **IBM Plex Sans Arabic Regular** (النص) · **Cairo SemiBold** (الواجهة)

تُحمَّل عبر `next/font` في `src/app/layout.js`، وتُطبَّق في آخر `src/app/globals.css`.

## الصفحات

- `/login` — دخول الطلاب وإنشاء الحساب واستعادة كلمة المرور
- `/` — الرئيسية، `/tasks`، `/about`، `/subject/...` — محمية بتسجيل الدخول
- `/admin` — لوحة الأدمن (نفس واجهة الدخول). الرابط يظهر بالضغط 5 مرات على شعار الفوتر.

## التشغيل محلياً

```bash
npm install
npm run dev
```

## منح صلاحية أدمن لحساب آخر

```sql

```

"use client";

import { useEffect, useState } from "react";
import Link from "next/link";
import { useApp } from "@/components/AppProvider";
import { SEMESTERS } from "@/data/semesters";
import { ALL_SUBJECTS } from "@/lib/subjects";

const FAV_KEY = "fav_subjects";

function loadFavs() {
  try {
    return JSON.parse(localStorage.getItem(FAV_KEY)) || [];
  } catch {
    return [];
  }
}

export default function HomePage() {
  const { L, lang } = useApp();
  const [semKey, setSemKey] = useState("semester3");
  const [favs, setFavs] = useState([]);

  useEffect(() => setFavs(loadFavs()), []);

  const toggleFav = (name) => {
    const list = loadFavs();
    const i = list.indexOf(name);
    if (i === -1) list.push(name);
    else list.splice(i, 1);
    try {
      localStorage.setItem(FAV_KEY, JSON.stringify(list));
    } catch {}
    setFavs(list);
  };

  const sem = SEMESTERS[semKey];
  const subjects = ALL_SUBJECTS.filter((s) => s.semester === semKey).sort(
    (a, b) => favs.includes(b.name) - favs.includes(a.name)
  );

  return (
    <div className="container">
      <div className="semester-tabs">
        <button type="button" className={"sem-btn" + (semKey === "semester3" ? " active" : "")}
          onClick={() => setSemKey("semester3")}>
          {L("📗 Semester 3", "📗 السمستر الثالث")}
        </button>
        <button type="button" className={"sem-btn" + (semKey === "semester2" ? " active" : "")}
          onClick={() => setSemKey("semester2")}>
          {L("📘 Semester 2", "📘 السمستر الثاني")}
        </button>
      </div>

      <div className="title">
        <h2>📚 {lang === "ar" ? sem.name.ar : sem.name.en}</h2>
      </div>
      <div className="counter">
        <span className="mono">{subjects.length}</span> {L("Academic Subjects", "مواد دراسية")}
      </div>

      <div className="subjects">
        {subjects.map((s) => {
          const fav = favs.includes(s.name);
          return (
            <Link key={s.slug} href={`/subject/${s.slug}/`} className="subject">
              <span
                className={"fav-star" + (fav ? " active" : "")}
                role="button"
                aria-label={L("Favorite", "تفضيل")}
                onClick={(e) => {
                  e.preventDefault();
                  e.stopPropagation();
                  toggleFav(s.name);
                }}
              >
                {fav ? "⭐" : "☆"}
              </span>
              <div className="icon">{s.icon}</div>
              <h3>{s.name}</h3>
            </Link>
          );
        })}
      </div>
    </div>
  );
}

"use client";

import { useEffect, useState } from "react";
import { useApp } from "@/components/AppProvider";

const TASKS_KEY = "study_tasks";

export default function TasksPage() {
  const { L } = useApp();
  const [tasks, setTasks] = useState([]);
  const [text, setText] = useState("");

  useEffect(() => {
    try {
      setTasks(JSON.parse(localStorage.getItem(TASKS_KEY)) || []);
    } catch {}
  }, []);

  const save = (list) => {
    setTasks(list);
    try {
      localStorage.setItem(TASKS_KEY, JSON.stringify(list));
    } catch (e) {
      console.error("Could not save tasks:", e);
    }
  };

  const add = () => {
    const v = text.trim();
    if (!v) return;
    save([...tasks, { text: v, done: false }]);
    setText("");
  };

  return (
    <div className="container">
      <div className="title">
        <h2>{L("📝 My Study Tasks", "📝 مهام مذاكرتي")}</h2>
      </div>
      <div className="task-add">
        <input
          type="text"
          value={text}
          onChange={(e) => setText(e.target.value)}
          onKeyDown={(e) => e.key === "Enter" && add()}
          placeholder={L("Write a new task...", "اكتب مهمة جديدة...")}
        />
        <button type="button" onClick={add}>{L("➕ Add", "➕ إضافة")}</button>
      </div>
      <ul className="task-list">
        {tasks.map((t, i) => (
          <li key={i} className={t.done ? "done" : ""}>
            <span onClick={() => save(tasks.map((x, j) => (j === i ? { ...x, done: !x.done } : x)))}>{t.text}</span>
            <button type="button" aria-label={L("Delete task", "حذف المهمة")}
              onClick={() => save(tasks.filter((_, j) => j !== i))}>✕</button>
          </li>
        ))}
      </ul>
      {tasks.length === 0 && (
        <p className="empty">{L("No tasks yet. Add your first one above 👆", "لا توجد مهام بعد. أضف أول مهمة من الأعلى 👆")}</p>
      )}
    </div>
  );
}

"use client";

import Link from "next/link";
import LoginShell from "@/components/LoginShell";
import LoginForm from "@/components/LoginForm";
import AdminDashboard from "@/components/AdminDashboard";
import { useApp } from "@/components/AppProvider";
import { useAuth } from "@/components/AuthProvider";

export default function AdminPage() {
  const { L } = useApp();
  const { session, loading, ready, isAdmin, recovery, signOut, user } = useAuth();

  if (loading || (session && !ready)) return <div className="splash" aria-busy="true">🔐</div>;

  // نفس واجهة تسجيل الدخول المستخدمة للطلاب
  if (!session || recovery) {
    return (
      <LoginShell>
        <LoginForm mode="admin" />
      </LoginShell>
    );
  }

  if (!isAdmin) {
    return (
      <LoginShell>
        <div className="lg-card">
          <div className="lg-head">
            <h1>{L("No admin access", "لا توجد صلاحية أدمن")}</h1>
            <p>
              {L(
                `${user?.email} is not an admin account. Ask the platform owner to grant access.`,
                `الحساب ${user?.email} ليس حساب أدمن. اطلب من مالك المنصة منحك الصلاحية.`
              )}
            </p>
          </div>
          <Link href="/" className="lg-submit lg-linkbtn">{L("Back to the platform", "العودة للمنصة")}</Link>
          <button type="button" className="lg-link strong" onClick={signOut}>{L("Sign out", "تسجيل خروج")}</button>
        </div>
      </LoginShell>
    );
  }

  return <AdminDashboard />;
}

"use client";

import { useEffect } from "react";
import { useRouter } from "next/navigation";
import LoginShell from "@/components/LoginShell";
import LoginForm from "@/components/LoginForm";
import { useAuth } from "@/components/AuthProvider";

export default function LoginPage() {
  const { session, loading, recovery } = useAuth();
  const router = useRouter();

  useEffect(() => {

  );
}

/** رابط التخزين العام -> مسار الملف داخل الـ bucket */
function storagePathFromUrl(url) {
  const marker = "/storage/v1/object/public/" + BUCKET + "/";
  const s = String(url || "");
  const i = s.indexOf(marker);
  if (i === -1) return null;
  try {
    return decodeURIComponent(s.slice(i + marker.length).split("?")[0]);
  } catch {
    return null;
  }
}

export default function AdminDashboard() {
  const { L, lang, toggleLang } = useApp();
  const { displayName, user, signOut } = useAuth();

  const [stats, setStats] = useState(null);
  const [pending, setPending] = useState(null);
  const [approved, setApproved] = useState(null);
  const [students, setStudents] = useState(null);
  const [studentQ, setStudentQ] = useState("");
  const [flashMsg, setFlashMsg] = useState("");

  const flash = (text) => {
    setFlashMsg(text);
    setTimeout(() => setFlashMsg(""), 8000);
  };

  const loadAll = useCallback(async () => {
    const [p, a, ps, as] = await Promise.all([
      supabase.from("submissions").select("*").eq("approved", false).order("created_at", { ascending: false }),
      supabase.from("submissions").select("*").eq("approved", true).order("created_at", { ascending: false }).limit(200),
      supabase.from("submissions").select("id", { count: "exact", head: true }).eq("approved", false),
      supabase.from("submissions").select("id", { count: "exact", head: true }).eq("approved", true),
    ]);
    setPending(p.error ? "error" : p.data || []);
    setApproved(a.error ? "error" : a.data || []);
    setStats(ps.error || as.error ? null : { pending: ps.count ?? 0, approved: as.count ?? 0 });
  }, []);

  const loadStudents = useCallback(async () => {
    const { data, error } = await supabase
      .from("profiles")
      .select("id,full_name,email,role,created_at")
      .order("created_at", { ascending: false })
      .limit(1000);
    setStudents(error ? "error" : data || []);
  }, []);

  useEffect(() => {
    loadAll();
    loadStudents();
  }, [loadAll, loadStudents]);

  const approve = async (id) => {
    const { error } = await supabase.from("submissions").update({ approved: true }).eq("id", id);
    if (error) {
      console.error(error);
      alert(L("Could not approve the file.", "تعذرت الموافقة."));
      return;
    }
    loadAll();
  };

  const remove = async (row) => {
    if (!confirm(L("Delete this file permanently?", "حذف هذا الملف نهائياً؟"))) return;
    let note = "";
    const filePath = storagePathFromUrl(row.path);
    if (filePath) {
      const { data, error } = await supabase.storage.from(BUCKET).remove([filePath]);
      if (error) {
        console.error(error);
        note = L("Could not delete the file from storage. Check the storage delete policy.", "تعذر حذف الملف من التخزين، تحقق من صلاحية الحذف على Storage.");

        note = L("File not found in storage (already deleted, or no delete permission).", "لم يُعثر على الملف في التخزين (ربما حُذف مسبقاً أو لا توجد صلاحية حذف).");
      }
    }
    const { error: dbErr } = await supabase.from("submissions").delete().eq("id", row.id);
    if (dbErr) {
      console.error(dbErr);
      alert(L("Could not delete.", "تعذر الحذف."));
      return;
    }
    if (note) flash(L("Record deleted. ", "تم حذف السجل من المنصة. ") + note);
    loadAll();
  };

  // البحث التلقائي عن الطلاب: يتحمل الأخطاء الإملائية، ويظهر للأدمن فقط
  const shownStudents = useMemo(() => {
    if (!Array.isArray(students)) return [];
    if (!studentQ.trim()) return students.slice(0, 50);
    return fuzzySearch(students, studentQ, (s) => [s.full_name, s.email], 50);
  }, [students, studentQ]);

  const empty = (t) => <div className="empty">{t}</div>;

    <div className="ad-page">
      <header className="ad-header">
        <h1>{L("🔐 Platform control panel", "🔐 لوحة تحكم المنصة")}</h1>
      </header>

      <div className="container ad-container">

            <div>{L("✅ Approved", "✅ موافق عليها")}</div>
          </div>
          <div className="ad-stat students">
            <div className="num mono">{Array.isArray(students) ? students.length : "–"}</div>
            <div>{L("🎓 Students", "🎓 الطلاب")}</div>
          </div>
        </div>

        <h2 className="ad-section">{L("⏳ Pending review", "⏳ بانتظار المراجعة")}</h2>
        {pending === null && empty(L("Loading...", "جارٍ التحميل..."))}
        {pending === "error" && empty(L("Could not load files. Make sure the submissions table exists and your account is an admin.", "تعذر تحميل الملفات. تأكد أن جدول submissions موجود وأن حسابك أدمن."))}

"use client";

import { createContext, useCallback, useContext, useEffect, useState } from "react";

const Ctx = createContext(null);
const LANG_KEY = "students_is_lang";
const DARK_KEY = "students_is_dark_mode";

export function AppProvider({ children }) {
  const [lang, setLang] = useState("en");
  const [dark, setDark] = useState(false);

  useEffect(() => {
    try {
      const l = localStorage.getItem(LANG_KEY);
      if (l === "ar" || l === "en") setLang(l);
      setDark(localStorage.getItem(DARK_KEY) === "1");
    } catch {}
  }, []);

  useEffect(() => {
    document.documentElement.lang = lang;
    document.documentElement.dir = lang === "ar" ? "rtl" : "ltr";
  }, [lang]);

  useEffect(() => {
    document.body.classList.toggle("dark", dark);
  }, [dark]);

  const toggleLang = () => {
    const next = lang === "en" ? "ar" : "en";
    setLang(next);
    try {
      localStorage.setItem(LANG_KEY, next);
    } catch {}
  };
  const toggleDark = () => {
    const next = !dark;
    setDark(next);
    try {
      localStorage.setItem(DARK_KEY, next ? "1" : "0");
    } catch {}
  };

  // L("English text", "النص العربي")
  const L = useCallback((en, ar) => (lang === "ar" ? ar : en), [lang]);

  return <Ctx.Provider value={{ lang, dark, toggleLang, toggleDark, L }}>{children}</Ctx.Provider>;
}

export const useApp = () => useContext(Ctx);

"use client";

import { createContext, useContext, useEffect, useState } from "react";
import { supabase } from "@/lib/supabase";

const Ctx = createContext(null);

export function AuthProvider({ children }) {
  const [session, setSession] = useState(null);
  const [loading, setLoading] = useState(true);
  const [recovery, setRecovery] = useState(false);
  // undefined = قيد التحميل، null = لا يوجد ملف شخصي
  const [profile, setProfile] = useState(undefined);

  useEffect(() => {
    let alive = true;
    supabase.auth
      .getSession()
      .then(({ data }) => {
        if (alive) setSession(data.session || null);
      })
      .catch(() => {})
      .finally(() => {
        if (alive) setLoading(false);
      });
    const { data: sub } = supabase.auth.onAuthStateChange((event, s) => {
      setSession(s || null);
      if (event === "PASSWORD_RECOVERY") setRecovery(true);
      if (event === "SIGNED_OUT") setRecovery(false);
    });
    return () => {
      alive = false;
      sub.subscription.unsubscribe();
    };
  }, []);

  const uid = session?.user?.id;
  useEffect(() => {
    if (!uid) {
      setProfile(undefined);
      return;
    }
    let alive = true;
    setProfile(undefined);
    supabase
      .from("profiles")
      .select("full_name,email,role")
      .eq("id", uid)
      .maybeSingle()
      .then(({ data, error }) => {
        if (alive) setProfile(error ? null : data || null);
      });
    return () => {
      alive = false;
    };
  }, [uid]);

  const user = session?.user || null;
  const value = {
    session,
    user,
    profile,
    loading,
    recovery,
    ready: !loading && (!session || profile !== undefined),
    isAdmin: profile?.role === "admin",
    displayName: profile?.full_name || user?.user_metadata?.full_name || user?.email || "",
    signIn: (email, password) => supabase.auth.signInWithPassword({ email, password }),
    signUp: (name, email, password) =>
      supabase.auth.signUp({ email, password, options: { data: { full_name: name } } }),
    signOut: () => supabase.auth.signOut(),
    sendReset: (email) =>
      supabase.auth.resetPasswordForEmail(email, { redirectTo: window.location.origin + "/login/" }),
    setNewPassword: async (password) => {
      const res = await supabase.auth.updateUser({ password });
      if (!res.error) setRecovery(false);
      return res;
    },
  };

  return <Ctx.Provider value={value}>{children}</Ctx.Provider>;
}

export const useAuth = () => useContext(Ctx);

"use client";

import { useEffect, useMemo, useRef, useState } from "react";
import { useRouter } from "next/navigation";
import { useApp } from "./AppProvider";
import { ALL_SUBJECTS } from "@/lib/subjects";
import { STATIC } from "@/data/files";
import { fuzzySearch } from "@/lib/search";
import { buildFileIndex } from "@/lib/files";

const subjectItems = ALL_SUBJECTS.map((s) => ({ type: "subject", ...s }));
const staticFileItems = ALL_SUBJECTS.flatMap((s) =>
  Object.entries(STATIC[s.name] || {}).flatMap(([material, files]) =>
    files.map((f) => ({ type: "file", name: f.name, url: f.url, subject: s.name, slug: s.slug, icon: s.icon, material }))
  )
);

/** بحث تلقائي: المواد + المحاضرات/الشيتات + الامتحانات، يتحمّل الأخطاء الإملائية */
export default function GlobalSearch() {
  const { L, lang } = useApp();
  const router = useRouter();
  const [q, setQ] = useState("");
  const [open, setOpen] = useState(false);
  const [active, setActive] = useState(0);
  const [files, setFiles] = useState(staticFileItems);
  const loaded = useRef(false);
  const box = useRef(null);

  // تحميل فهرس الملفات الكامل عند أول تركيز على مربع البحث
  const warm = () => {
    if (loaded.current) return;
    loaded.current = true;
    buildFileIndex()
      .then((idx) => {
        const seen = new Set();
        const merged = [...staticFileItems, ...idx].filter((f) => {
          const k = f.url + "|" + f.material;
          if (seen.has(k)) return false;
          seen.add(k);
          return true;
        });
        setFiles(merged);
      })
      .catch(() => {});
  };

  const results = useMemo(() => {
    if (!q.trim()) return [];
    const subs = fuzzySearch(subjectItems, q, (s) => [s.name, ...s.aliases], 5);
    const fls = fuzzySearch(files, q, (f) => [f.name, f.subject], 6);
    return [...subs, ...fls];
  }, [q, files]);

  useEffect(() => setActive(0), [q]);
  useEffect(() => {
    const onDoc = (e) => {
      if (box.current && !box.current.contains(e.target)) setOpen(false);
    };
    document.addEventListener("mousedown", onDoc);
    return () => document.removeEventListener("mousedown", onDoc);
  }, []);

  const choose = (item) => {
    setOpen(false);
    setQ("");
    if (item.type === "subject") router.push(`/subject/${item.slug}/`);
    else router.push(`/subject/${item.slug}/${item.material}/?q=${encodeURIComponent(item.name)}`);
  };

  const onKey = (e) => {
    if (e.key === "ArrowDown") {
      e.preventDefault();
      setOpen(true);
      setActive((a) => Math.min(a + 1, results.length - 1));
    } else if (e.key === "ArrowUp") {
      e.preventDefault();
      setActive((a) => Math.max(a - 1, 0));
    } else if (e.key === "Enter" && results[active]) {
      e.preventDefault();
      choose(results[active]);
    } else if (e.key === "Escape") {
      setOpen(false);
    }
  };

  const materialLabel = (m) =>
    m === "Exams" ? L("Exam", "امتحان") : L("Lecture / Sheet", "محاضرة / شيت");

  return (
    <div className="search" ref={box}>
      <input
        type="search"
        role="combobox"
        aria-expanded={open && results.length > 0}
        aria-controls="gs-list"
        autoComplete="off"
        value={q}
        placeholder={L("🔎 Search subjects, lectures, sheets...", "🔎 ابحث عن مادة أو محاضرة أو شيت...")}
        onFocus={() => {
          warm();
          setOpen(true);
        }}
        onChange={(e) => {
          setQ(e.target.value);
          setOpen(true);
        }}
        onKeyDown={onKey}
      />
      {open && q.trim() && (
        <ul className="gs-list" id="gs-list" role="listbox">
          {results.length === 0 && (
            <li className="gs-empty">{L("No matches. Try fewer letters.", "لا توجد نتائج. جرّب كلمة أقصر.")}</li>
          )}
          {results.map((item, i) => (
            <li
              key={item.type + (item.url || item.slug) + i}
              role="option"
              aria-selected={i === active}
              className={"gs-item" + (i === active ? " active" : "")}
              onMouseEnter={() => setActive(i)}
              onMouseDown={(e) => {
                e.preventDefault();
                choose(item);
              }}
            >
              <span className="gs-ico">{item.icon}</span>
              <span className="gs-text">
                <b>{item.name}</b>
                <small>
                  {item.type === "subject"
                    ? L("Subject", "مادة")
                    : `${item.subject} · ${materialLabel(item.material)}`}
                </small>
              </span>
            </li>
          ))}
        </ul>
      )}
    </div>
  );
}

function friendly(err, L) {
  const m = String(err?.message || "").toLowerCase();
  if (m.includes("invalid login"))
    return L("Email or password is incorrect.", "البريد الإلكتروني أو كلمة المرور غير صحيحة.");
  if (m.includes("email not confirmed"))
    return L("Confirm your email first, then sign in.", "أكّد بريدك الإلكتروني أولاً ثم سجّل الدخول.");
  if (m.includes("already registered") || m.includes("already been registered"))
    return L("This email already has an account. Sign in instead.", "هذا البريد لديه حساب بالفعل. سجّل الدخول بدلاً من ذلك.");
  if (m.includes("password") && (m.includes("least") || m.includes("short") || m.includes("weak")))
    return L("Password must be at least 6 characters.", "يجب ألا تقل كلمة المرور عن 6 أحرف.");
  if (m.includes("rate limit") || m.includes("too many"))
    return L("Too many attempts. Wait a minute and try again.", "محاولات كثيرة. انتظر دقيقة ثم حاول مجدداً.");
  if (m.includes("failed to fetch") || m.includes("network"))
    return L("No connection. Check your internet and try again.", "لا يوجد اتصال. تحقق من الإنترنت وحاول مجدداً.");
  return L("Something went wrong. Try again.", "حدث خطأ. حاول مجدداً.");
}

function Mcq({ subject }) {
  const { L, lang } = useApp();
  const list = QUESTIONS_DATA[subject.name] || [];
  const [picked, setPicked] = useState({}); // {questionIndex: optionIndex}

  if (!list.length) return <div className="empty">📂 {L("No questions available.", "لا توجد أسئلة متاحة حالياً.")}</div>;

  return (
    <div className="mcq-wrap">

            <th>#</th>
            <th>{L("Question", "السؤال")}</th>
            <th>{L("Options", "الخيارات")}</th>
          </tr>
        </thead>
        <tbody>
          {list.map((item, qi) => {
            const opts = item.options[lang] || item.options.en || item.options.ar;
            const chosen = picked[qi];
            return (
              <tr key={qi}>
                <td className="mono" style={{ fontWeight: 700, width: 50 }}>{qi + 1}</td>
                <td style={{ fontWeight: 700 }}>{item.q[lang] || item.q.en || item.q.ar}</td>
                <td>
                  {opts.map((opt, oi) => {
                    let cls = "option-btn";
                    if (chosen !== undefined) {
                      if (oi === item.correct) cls += " correct";
                      else if (oi === chosen) cls += " wrong";
                    }
                    return (
                      <button key={oi} type="button" className={cls} disabled={chosen !== undefined}
                        onClick={() => setPicked((p) => ({ ...p, [qi]: oi }))}>
                        {opt}
                      </button>
                    );
                  })}
                </td>
              </tr>
            );
          })}
        </tbody>
      </table>
    </div>
  );
}

function Files({ subject, material }) {
  const { L } = useApp();
  const [files, setFiles] = useState(null); // null = loading, "error"
  const [q, setQ] = useState("");
  const [msg, setMsg] = useState(null); // {text, err}
  const [busy, setBusy] = useState(false);
  const fileInput = useRef(null);

  useEffect(() => {
    let alive = true;
    setFiles(null);
    loadFiles(subject.name, material)
      .then((d) => alive && setFiles(d))
      .catch((e) => {
        console.error(e);
        if (alive) setFiles("error");
      });
    return () => {
      alive = false;
    };
  }, [subject.name, material]);

  // فتح الصفحة من نتيجة البحث التلقائي: ?q=اسم الملف
  useEffect(() => {
    try {
      const v = new URLSearchParams(window.location.search).get("q");
      if (v) setQ(v);
    } catch {}
  }, []);

  const flash = (text, err) => {
    setMsg({ text, err });
    setTimeout(() => setMsg(null), 6000);
  };

  const onUpload = async () => {
    const file = fileInput.current?.files?.[0];
    if (!file || busy) return;
    setBusy(true);
    flash(L("Uploading...", "جارٍ الرفع..."), false);
    try {
      await uploadSubmission(subject.name, material, file);
      flash(L("Uploaded. It will appear after admin approval ✅", "تم الرفع، سيظهر بعد موافقة الأدمن ✅"), false);
      fileInput.current.value = "";
    } catch (e) {
      console.error(e);
      flash(L("Could not upload the file.", "تعذر رفع الملف."), true);
    } finally {
      setBusy(false);
    }
  };

  let shown = Array.isArray(files) ? files : [];
  if (q.trim()) {
    const direct = shown.filter((f) => norm(f.name).includes(norm(q)));
    shown = direct.length ? direct : fuzzySearch(shown, q, (f) => [f.name], 50);
  }

  return (
    <>
      <div className="upload-card">
        <h3>{L("📤 Upload a File (reviewed before publishing)", "📤 رفع ملف (يُنشر بعد المراجعة)")}</h3>
        <input type="file" ref={fileInput} accept="application/pdf,.doc,.docx,.ppt,.pptx,image/*" />
        <button type="button" onClick={onUpload} disabled={busy}>{L("Upload", "رفع")}</button>
        {msg && <div className={"msg show" + (msg.err ? " err" : "")}>{msg.text}</div>}
      </div>

      <input type="text" className="file-search" value={q} onChange={(e) => setQ(e.target.value)}
        placeholder={L("🔎 Search files in this section...", "🔎 ابحث في هذا القسم...")} />

      <div className="file-list">
        {files === null && <div className="empty">{L("Loading...", "جارٍ التحميل...")}</div>}
        {files === "error" && <div className="empty">{L("Could not load files.", "تعذر تحميل الملفات.")}</div>}
        {Array.isArray(files) && files.length === 0 && (
          <div className="empty">📂 {L("No files available yet.", "لا توجد ملفات بعد.")}</div>
        )}
        {Array.isArray(files) && files.length > 0 && shown.length === 0 && (
          <div className="empty">{L("No files match your search.", "لا توجد ملفات مطابقة للبحث.")}</div>
        )}
        {shown.map((f) => (
          <div className="file" key={f.url}>
            <span>📄 {f.name}</span>
            <button type="button" onClick={() => window.open(f.url, "_blank", "noopener")}>
              {L("Open", "فتح")}
            </button>
          </div>
        ))}
      </div>
    </>
  );
}

"use client";

import { useEffect, useState } from "react";
import Link from "next/link";
import { useApp } from "./AppProvider";
import { findSubject } from "@/lib/subjects";
import { MATERIALS } from "@/data/semesters";

const EXAM_KEY = "exam_dates";

function loadDates() {
  try {
    return JSON.parse(localStorage.getItem(EXAM_KEY)) || {};
  } catch {
    return {};
  }
}

export default function SubjectView({ slug }) {
  const { L, lang } = useApp();
  const subject = findSubject(slug);
  const [dateStr, setDateStr] = useState("");
  const [saved, setSaved] = useState("");

  useEffect(() => {
    if (!subject) return;
    const v = loadDates()[subject.name] || "";
    setSaved(v);
    setDateStr(v);
  }, [subject]);

  if (!subject) {
    return (
      <div className="container">
        <p className="empty">{L("Subject not found.", "المادة غير موجودة.")}</p>
        <Link href="/" className="back">{L("← Back", "→ رجوع")}</Link>
      </div>
    );
  }

  const persist = (value) => {
    const dates = loadDates();
    if (value) dates[subject.name] = value;
    else delete dates[subject.name];
    try {
      localStorage.setItem(EXAM_KEY, JSON.stringify(dates));
    } catch (e) {
      console.error("Could not save exam date:", e);
    }
    setSaved(value);
  };

  // العد التنازلي للامتحان
  let examNode = L("⏳ No exam date set", "⏳ لم يُحدد موعد امتحان");
  let urgent = false;
  if (saved) {
    const today = new Date();
    today.setHours(0, 0, 0, 0);
    const days = Math.round((new Date(saved + "T00:00:00") - today) / 86400000);
    urgent = days >= 0 && days <= 2;
    if (days < 0) examNode = L("✅ This exam date has passed", "✅ انتهى موعد هذا الامتحان");
    else if (days === 0) examNode = L("🚨 Exam is today!", "🚨 الامتحان اليوم!");
    else
      examNode =
        lang === "ar" ? (
          <>⏳ باقي <span className="mono">{days}</span> يوم على الامتحان</>
        ) : (
          <>⏳ <span className="mono">{days}</span> day(s) left until the exam</>
        );
  }

  return (
    <div className="container">
      <Link href="/" className="back">{L("← Back", "→ رجوع")}</Link>
      <div className="subject-title">
        <h2>📚 {subject.name}</h2>
      </div>

      <div className={"exam-card" + (urgent ? " urgent" : "")}>
        <span>{examNode}</span>
        <input type="date" value={dateStr} onChange={(e) => setDateStr(e.target.value)} />
        <button type="button" onClick={() => dateStr && persist(dateStr)}>{L("Save", "حفظ")}</button>
        <button type="button" className="clear-btn" onClick={() => { setDateStr(""); persist(""); }}>
          {L("Clear", "مسح")}
        </button>
      </div>

      <div className="materials">
        {MATERIALS.map(([key, icon, en, ar]) => (
          <Link key={key} href={`/subject/${subject.slug}/${key}/`} className="material">
            <div className="material-icon">{icon}</div>
            <h3>{lang === "ar" ? ar : key}</h3>
            <p>{lang === "ar" ? ar : en}</p>
          </Link>
        ))}
      </div>
    </div>
  );
}

import { supabase, BUCKET } from "@/lib/supabase";
import { STATIC } from "@/data/files";
import { ALL_SUBJECTS, folderOf } from "@/lib/subjects";

const cache = new Map();

async function listFiles(path) {
  const { data, error } = await supabase.storage
    .from(BUCKET)
    .list(path, { limit: 200, sortBy: { column: "name", order: "asc" } });
  if (error) throw error;
  return (data || [])
    .filter((f) => f.id && f.name !== ".emptyFolderPlaceholder")
    .map((f) => ({
      name: f.name.replace(/\.[^.]+$/, ""),
      url: supabase.storage.from(BUCKET).getPublicUrl(path + "/" + f.name).data.publicUrl,
    }));
}

/** ملفات مادة/قسم: الروابط الثابتة + ملفات التخزين + الملفات المعتمدة من الطلاب */
export async function loadFiles(subjectName, material) {
  const key = subjectName + "|" + material;
  if (cache.has(key)) return cache.get(key);

  const fixed = (STATIC[subjectName] && STATIC[subjectName][material]) || [];
  let listed = [];
  let approved = [];

  try {
    const folder = folderOf(subjectName);
    listed = await listFiles(folder + "/" + material);
    if (material === "Lectures") listed = listed.concat(await listFiles(folder));
  } catch (e) {
    console.error(e);
    if (!fixed.length) throw e;
  }
  try {
    const { data, error } = await supabase
      .from("submissions")
      .select("name,path")
      .eq("subject", subjectName)
      .eq("material", material)
      .eq("approved", true);
    if (!error) approved = (data || []).map((r) => ({ name: r.name, url: r.path }));
  } catch (e) {
    console.error("Could not load approved submissions:", e);
  }

  const seen = new Set();
  const out = [];
  fixed.concat(listed, approved).forEach((f) => {
    if (!seen.has(f.url)) {
      seen.add(f.url);
      out.push(f);
    }
  });
  cache.set(key, out);
  return out;
}

/** رفع ملف من طالب (يظهر بعد موافقة الأدمن) */
export async function uploadSubmission(subject, material, file) {
  const safeName = file.name.replace(/[^\w.\-]+/g, "_");
  const path = "pending/" + Date.now() + "_" + safeName;
  const { error: upErr } = await supabase.storage.from(BUCKET).upload(path, file);
  if (upErr) throw upErr;
  const url = supabase.storage.from(BUCKET).getPublicUrl(path).data.publicUrl;
  const { error: insErr } = await supabase.from("submissions").insert([
    {
      subject,
      material,
      name: file.name.replace(/\.[^.]+$/, "").slice(0, 120),
      path: url,
      approved: false,
    },
  ]);
  if (insErr) throw insErr;
}

/**
 * فهرس البحث للملفات (محاضرات وامتحانات كل المواد).
 * يبدأ بالروابط الثابتة فوراً، ثم يكمّل من التخزين بالتوازي المحدود.
 */
let indexPromise = null;
export function buildFileIndex() {
  if (indexPromise) return indexPromise;
  indexPromise = (async () => {
    const jobs = [];
    for (const s of ALL_SUBJECTS) {
      for (const material of ["Lectures", "Exams"]) jobs.push({ s, material });
    }
    const out = [];
    let i = 0;
    async function worker() {
      while (i < jobs.length) {
        const { s, material } = jobs[i++];
        try {
          const files = await loadFiles(s.name, material);
          files.forEach((f) =>
            out.push({ type: "file", name: f.name, url: f.url, subject: s.name, slug: s.slug, icon: s.icon, material })
          );
        } catch {
          /* تجاهل مادة تعذّر تحميلها */
        }
      }
    }
    await Promise.all([worker(), worker(), worker(), worker(), worker()]);
    return out;
  })();
  return indexPromise;
}

// بحث تلقائي يتحمّل الأخطاء الإملائية، ويدعم العربي والإنجليزي.

const ROMAN = { i: "1", ii: "2", iii: "3", iv: "4" };

/** توحيد النص: حروف صغيرة، حذف التشكيل، توحيد الهمزات والتاء المربوطة والألف المقصورة، وتحويل الأرقام العربية */
export function norm(input) {
  return String(input || "")
    .toLowerCase()
    .replace(/[\u064B-\u065F\u0670\u0640]/g, "")
    .replace(/[أإآٱ]/g, "ا")
    .replace(/ى/g, "ي")
    .replace(/ة/g, "ه")
    .replace(/ؤ/g, "و")
    .replace(/ئ/g, "ي")
    .replace(/[٠-٩]/g, (d) => String("٠١٢٣٤٥٦٧٨٩".indexOf(d)))
    .replace(/[^\p{L}\p{N}\s]/gu, " ")
    .replace(/\s+/g, " ")
    .trim();
}

export function tokens(input) {
  return norm(input)
    .split(" ")
    .filter(Boolean)
    .map((t) => ROMAN[t] || t);
}

/** مسافة التحرير (Levenshtein) مع توقف مبكر */
function lev(a, b, max) {
  if (Math.abs(a.length - b.length) > max) return max + 1;
  const prev = new Array(b.length + 1);
  for (let j = 0; j <= b.length; j++) prev[j] = j;
  for (let i = 1; i <= a.length; i++) {
    let last = prev[0];
    prev[0] = i;
    let rowMin = prev[0];
    for (let j = 1; j <= b.length; j++) {
      const tmp = prev[j];
      prev[j] = Math.min(
        prev[j] + 1,
        prev[j - 1] + 1,
        last + (a[i - 1] === b[j - 1] ? 0 : 1)
      );
      last = tmp;
      if (prev[j] < rowMin) rowMin = prev[j];
    }
    if (rowMin > max) return max + 1;
  }
  return prev[b.length];
}

function tokenScore(q, t) {
  if (t === q) return 100;
  if (t.startsWith(q)) return 90 - Math.min(10, t.length - q.length);
  if (q.length >= 3 && t.includes(q)) return 70;
  if (q.length >= 3) {
    const max = q.length <= 4 ? 1 : q.length <= 8 ? 2 : 3;
    const d = Math.min(lev(q, t.slice(0, q.length), max), lev(q, t, max));
    if (d <= max) return 60 - d * 10;
  }
  return 0;
}

/** تقييم عنصر مقابل الاستعلام: 0 = غير مطابق */
export function scoreText(queryTokens, text) {
  const tt = tokens(text);
  if (!tt.length || !queryTokens.length) return 0;
  let total = 0;
  for (const q of queryTokens) {
    let best = 0;
    for (const t of tt) {
      const s = tokenScore(q, t);
      if (s > best) best = s;
    }
    if (!best) {
      // الطالب كتب الكلمات ملتصقة؟ جرّب مطابقتها مع النص كاملاً بلا مسافات
      const joined = tt.join("");
      if (q.length >= 4 && joined.includes(q)) best = 55;
    }
    if (!best) return 0;
    total += best;
  }
  return total / queryTokens.length;
}

/**
 * items: مصفوفة عناصر، getTexts(item) => مصفوفة نصوص قابلة للبحث
 * يرجع أفضل النتائج مرتبة.
 */
export function fuzzySearch(items, query, getTexts, limit = 8) {
  const q = tokens(query);
  if (!q.length) return [];
  const scored = [];
  for (const item of items) {
    let best = 0;
    for (const text of getTexts(item)) {
      const s = scoreText(q, text);
      if (s > best) best = s;
    }
    if (best > 0) scored.push({ item, score: best });
  }
  scored.sort((a, b) => b.score - a.score);
  return scored.slice(0, limit).map((x) => x.item);
}
